EXPERIMENT 1: Feedforward Neural Network (MNIST)
--------------------------------------------------------------------------------
import tensorflow as tf
from tensorflow import keras
from tensorflow.keras import layers
import numpy as np
import matplotlib.pyplot as plt

# 1. Load and normalize dataset
(x_train, y_train), (x_test, y_test) = keras.datasets.mnist.load_data()
x_train = x_train.astype("float32") / 255.0
x_test = x_test.astype("float32") / 255.0

# 2. Build Feedforward Model
model = keras.Sequential([
    keras.Input(shape=(28, 28)),
    layers.Flatten(),
    layers.Dense(128, activation="relu"),
    layers.Dense(64, activation="relu"),
    layers.Dense(10, activation="softmax")
])

# 3. Compile Model
model.compile(
    optimizer="adam",
    loss="sparse_categorical_crossentropy",
    metrics=["accuracy"]
)

# 4. Train Model
history = model.fit(
    x_train,
    y_train,
    epochs=10,
    batch_size=32,
    validation_split=0.1
)

# 5. Evaluate
test_loss, test_accuracy = model.evaluate(x_test, y_test, verbose=0)
print(f"\nTest Loss: {test_loss:.4f} | Test Accuracy: {test_accuracy * 100:.2f}%")

# 6. Plot Accuracy and Loss
epochs_range = range(1, len(history.history["accuracy"]) + 1)
plt.figure(figsize=(10, 4))

plt.subplot(1, 2, 1)
plt.plot(epochs_range, history.history["accuracy"], label="Train Acc")
plt.plot(epochs_range, history.history["val_accuracy"], label="Val Acc")
plt.title("Accuracy")
plt.xlabel("Epoch")
plt.ylabel("Accuracy")
plt.legend()
plt.grid(True)

plt.subplot(1, 2, 2)
plt.plot(epochs_range, history.history["loss"], label="Train Loss")
plt.plot(epochs_range, history.history["val_loss"], label="Val Loss")
plt.title("Loss")
plt.xlabel("Epoch")
plt.ylabel("Loss")
plt.legend()
plt.grid(True)

plt.tight_layout()
plt.show()

EXPERIMENT 2: Activation Functions & Derivatives Comparison
--------------------------------------------------------------------------------
import numpy as np
import matplotlib.pyplot as plt

def relu(x):
    return np.maximum(0, x)

def relu_derivative(x):
    return np.where(x > 0, 1, 0)

def sigmoid(x):
    return 1 / (1 + np.exp(-x))

def sigmoid_derivative(x):
    s = sigmoid(x)
    return s * (1 - s)

def tanh(x):
    return np.tanh(x)

def tanh_derivative(x):
    return 1 - np.tanh(x) ** 2

x = np.linspace(-10, 10, 400)

plt.figure(figsize=(14, 6))

plt.subplot(2, 3, 1)
plt.plot(x, relu(x), color="tab:blue")
plt.title("ReLU")
plt.grid(True)

plt.subplot(2, 3, 2)
plt.plot(x, sigmoid(x), color="tab:orange")
plt.title("Sigmoid")
plt.grid(True)

plt.subplot(2, 3, 3)
plt.plot(x, tanh(x), color="tab:green")
plt.title("Tanh")
plt.grid(True)

plt.subplot(2, 3, 4)
plt.plot(x, relu_derivative(x), color="tab:red")
plt.title("ReLU Derivative")
plt.grid(True)

plt.subplot(2, 3, 5)
plt.plot(x, sigmoid_derivative(x), color="tab:purple")
plt.title("Sigmoid Derivative")
plt.grid(True)

plt.subplot(2, 3, 6)
plt.plot(x, tanh_derivative(x), color="tab:brown")
plt.title("Tanh Derivative")
plt.grid(True)

EXPERIMENT 3: Multi-Layer Perceptron (MLP) with Dropout
--------------------------------------------------------------------------------
import tensorflow as tf
from tensorflow.keras import layers, models
import matplotlib.pyplot as plt
import numpy as np

# 1. Load and flatten dataset
(x_train, y_train), (x_test, y_test) = tf.keras.datasets.mnist.load_data()
x_train = x_train.reshape(-1, 784).astype("float32") / 255.0
x_test = x_test.reshape(-1, 784).astype("float32") / 255.0

# 2. Build MLP Model
model = models.Sequential([
    layers.Input(shape=(784,)),
    layers.Dense(256, activation="relu"),
    layers.Dropout(0.2),
    layers.Dense(128, activation="relu"),
    layers.Dropout(0.2),
    layers.Dense(10, activation="softmax")
])

# 3. Compile and fit
model.compile(
    optimizer="adam",
    loss="sparse_categorical_crossentropy",
    metrics=["accuracy"]
)
model.summary()

history = model.fit(x_train, y_train, epochs=10, batch_size=128, validation_split=0.1)

# 4. Evaluate
test_loss, test_acc = model.evaluate(x_test, y_test, verbose=0)
print(f"\nTest Accuracy: {test_acc:.4f} | Test Loss: {test_loss:.4f}")

# 5. Visualize Predictions
preds = np.argmax(model.predict(x_test[:10]), axis=1)
fig, axes = plt.subplots(2, 5, figsize=(10, 4))
for i, ax in enumerate(axes.flat):
    ax.imshow(x_test[i].reshape(28, 28), cmap="gray")
    ax.set_title(f"P: {preds[i]} | T: {y_test[i]}")
    ax.axis("off")

plt.tight_layout()
plt.show()

EXPERIMENT 4: Loss Function Exploration (Cross-Entropy vs MSE)
--------------------------------------------------------------------------------
import matplotlib.pyplot as plt
import tensorflow as tf
from tensorflow.keras import layers, models

# 1. Load and preprocess data
(x_train, y_train), (x_test, y_test) = tf.keras.datasets.mnist.load_data()
x_train = x_train.reshape(-1, 784).astype("float32") / 255.0
x_test = x_test.reshape(-1, 784).astype("float32") / 255.0

# One-hot encoding for categorical crossentropy and MSE
y_train_cat = tf.keras.utils.to_categorical(y_train, 10)
y_test_cat = tf.keras.utils.to_categorical(y_test, 10)

def build_model():
    return models.Sequential([
        layers.Input(shape=(784,)),
        layers.Dense(128, activation="relu"),
        layers.Dense(10, activation="softmax")
    ])

# 2. Train with Categorical Cross-Entropy
print("\n--- Training with Categorical Cross-Entropy ---")
model_ce = build_model()
model_ce.compile(optimizer="adam", loss="categorical_crossentropy", metrics=["accuracy"])
history_ce = model_ce.fit(x_train, y_train_cat, epochs=5, batch_size=64, validation_split=0.1, verbose=1)

# 3. Train with Mean Squared Error
print("\n--- Training with Mean Squared Error (MSE) ---")
model_mse = build_model()
model_mse.compile(optimizer="adam", loss="mean_squared_error", metrics=["accuracy"])
history_mse = model_mse.fit(x_train, y_train_cat, epochs=5, batch_size=64, validation_split=0.1, verbose=1)

# 4. Compare Results
loss_ce, acc_ce = model_ce.evaluate(x_test, y_test_cat, verbose=0)
loss_mse, acc_mse = model_mse.evaluate(x_test, y_test_cat, verbose=0)
print(f"\nCE Loss  -> Test Loss: {loss_ce:.4f} | Test Acc: {acc_ce:.4f}")
print(f"MSE Loss -> Test Loss: {loss_mse:.4f} | Test Acc: {acc_mse:.4f}")

# 5. Plot Comparison
plt.figure(figsize=(10, 4))
plt.subplot(1, 2, 1)
plt.plot(history_ce.history["val_accuracy"], label="CE Val Acc")
plt.plot(history_mse.history["val_accuracy"], label="MSE Val Acc")
plt.title("Validation Accuracy")
plt.xlabel("Epoch")
plt.ylabel("Accuracy")
plt.legend()
plt.grid(True)

plt.subplot(1, 2, 2)
plt.plot(history_ce.history["val_loss"], label="CE Val Loss")
plt.plot(history_mse.history["val_loss"], label="MSE Val Loss")
plt.title("Validation Loss")
plt.xlabel("Epoch")
plt.ylabel("Loss")
plt.legend()
plt.grid(True)

plt.tight_layout()
plt.show()

EXPERIMENT 5: Gradient Descent Optimization (SGD, Adam, RMSprop)
--------------------------------------------------------------------------------
import matplotlib.pyplot as plt
import tensorflow as tf
from tensorflow.keras import layers, models

(x_train, y_train), (x_test, y_test) = tf.keras.datasets.mnist.load_data()
x_train = x_train.reshape(-1, 784).astype("float32") / 255.0
x_test = x_test.reshape(-1, 784).astype("float32") / 255.0

def build_model():
    return models.Sequential([
        layers.Input(shape=(784,)),
        layers.Dense(128, activation="relu"),
        layers.Dense(10, activation="softmax")
    ])

optimizers = {
    "SGD": tf.keras.optimizers.SGD(learning_rate=0.01, momentum=0.9),
    "RMSprop": tf.keras.optimizers.RMSprop(learning_rate=0.001),
    "Adam": tf.keras.optimizers.Adam(learning_rate=0.001)
}

histories = {}

for name, opt in optimizers.items():
    print(f"\nTraining with optimizer: {name}")
    model = build_model()
    model.compile(
        optimizer=opt,
        loss="sparse_categorical_crossentropy",
        metrics=["accuracy"]
    )
    hist = model.fit(x_train, y_train, epochs=8, batch_size=128, validation_split=0.1, verbose=0)
    histories[name] = hist
    test_loss, test_acc = model.evaluate(x_test, y_test, verbose=0)
    print(f"{name} -> Test Accuracy: {test_acc:.4f}, Test Loss: {test_loss:.4f}")

plt.figure(figsize=(10, 4))
plt.subplot(1, 2, 1)
for name, hist in histories.items():
    plt.plot(hist.history["val_accuracy"], label=name)
plt.title("Val Accuracy by Optimizer")
plt.xlabel("Epoch")
plt.ylabel("Accuracy")
plt.legend()
plt.grid(True)

plt.subplot(1, 2, 2)
for name, hist in histories.items():
    plt.plot(hist.history["val_loss"], label=name)
plt.title("Val Loss by Optimizer")
plt.xlabel("Epoch")
plt.ylabel("Loss")
plt.legend()
plt.grid(True)

plt.tight_layout()
plt.show()

EXPERIMENT 6: CNN from Scratch (CIFAR-10)
--------------------------------------------------------------------------------
import tensorflow as tf
from tensorflow.keras import layers, models
import matplotlib.pyplot as plt

# 1. Load and scale data
(x_train, y_train), (x_test, y_test) = tf.keras.datasets.cifar10.load_data()
x_train = x_train.astype("float32") / 255.0
x_test = x_test.astype("float32") / 255.0
y_train = y_train.flatten()
y_test = y_test.flatten()

# 2. Build CNN Architecture
model = models.Sequential([
    layers.Input(shape=(32, 32, 3)),
    layers.Conv2D(32, (3, 3), activation="relu", padding="same"),
    layers.BatchNormalization(),
    layers.Conv2D(32, (3, 3), activation="relu", padding="same"),
    layers.MaxPooling2D((2, 2)),
    layers.Dropout(0.25),

    layers.Conv2D(64, (3, 3), activation="relu", padding="same"),
    layers.BatchNormalization(),
    layers.Conv2D(64, (3, 3), activation="relu", padding="same"),
    layers.MaxPooling2D((2, 2)),
    layers.Dropout(0.25),

    layers.Flatten(),
    layers.Dense(256, activation="relu"),
    layers.Dropout(0.5),
    layers.Dense(10, activation="softmax")
])

model.compile(
    optimizer="adam",
    loss="sparse_categorical_crossentropy",
    metrics=["accuracy"]
)
model.summary()

# 3. Train
history = model.fit(x_train, y_train, epochs=15, batch_size=64, validation_split=0.1)

# 4. Evaluate
test_loss, test_acc = model.evaluate(x_test, y_test, verbose=0)
print(f"\nTest Accuracy: {test_acc:.4f} | Test Loss: {test_loss:.4f}")

# 5. Plot Curves
plt.figure(figsize=(10, 4))
plt.subplot(1, 2, 1)
plt.plot(history.history["accuracy"], label="Train")
plt.plot(history.history["val_accuracy"], label="Val")
plt.title("CNN Accuracy")
plt.legend()
plt.grid(True)

plt.subplot(1, 2, 2)
plt.plot(history.history["loss"], label="Train")
plt.plot(history.history["val_loss"], label="Val")
plt.title("CNN Loss")
plt.legend()
plt.grid(True)

plt.tight_layout()
plt.show()

EXPERIMENT 7: Transfer Learning using ResNet50 (CIFAR-10)
--------------------------------------------------------------------------------
import tensorflow as tf
from tensorflow.keras import layers, models
from tensorflow.keras.applications import ResNet50
from tensorflow.keras.applications.resnet50 import preprocess_input
import matplotlib.pyplot as plt

IMG_SIZE = 96

# 1. Load data subset for practical runtime
(x_train, y_train), (x_test, y_test) = tf.keras.datasets.cifar10.load_data()
y_train, y_test = y_train.flatten(), y_test.flatten()

x_train, y_train = x_train[:10000], y_train[:10000]
x_test, y_test = x_test[:2000], y_test[:2000]

def preprocess(images, labels):
    images = tf.image.resize(images, (IMG_SIZE, IMG_SIZE))
    images = preprocess_input(images)
    return images, labels

train_ds = tf.data.Dataset.from_tensor_slices((x_train, y_train)) \
    .map(preprocess).shuffle(1000).batch(32).prefetch(tf.data.AUTOTUNE)

test_ds = tf.data.Dataset.from_tensor_slices((x_test, y_test)) \
    .map(preprocess).batch(32).prefetch(tf.data.AUTOTUNE)

# 2. Base Model Setup
base_model = ResNet50(weights="imagenet", include_top=False, input_shape=(IMG_SIZE, IMG_SIZE, 3))
base_model.trainable = False

# 3. Add Custom Head
model = models.Sequential([
    base_model,
    layers.GlobalAveragePooling2D(),
    layers.Dense(128, activation="relu"),
    layers.Dropout(0.3),
    layers.Dense(10, activation="softmax")
])

model.compile(optimizer=tf.keras.optimizers.Adam(1e-3),
              loss="sparse_categorical_crossentropy",
              metrics=["accuracy"])

# 4. Phase 1: Feature Extraction
history = model.fit(train_ds, validation_data=test_ds, epochs=5)

# 5. Phase 2: Fine-Tuning
base_model.trainable = True
for layer in base_model.layers[:-20]:
    layer.trainable = False

model.compile(optimizer=tf.keras.optimizers.Adam(1e-5),
              loss="sparse_categorical_crossentropy",
              metrics=["accuracy"])

history_fine = model.fit(train_ds, validation_data=test_ds, epochs=5)

# 6. Evaluation
test_loss, test_acc = model.evaluate(test_ds, verbose=0)
print(f"\nFinal Test Accuracy: {test_acc:.4f} | Test Loss: {test_loss:.4f}")

# 7. Plotting
acc = history.history["accuracy"] + history_fine.history["accuracy"]
val_acc = history.history["val_accuracy"] + history_fine.history["val_accuracy"]
plt.figure(figsize=(7, 4))
plt.plot(acc, label="Train Accuracy")
plt.plot(val_acc, label="Val Accuracy")
plt.axvline(x=len(history.history["accuracy"]) - 1, color="gray", linestyle="--", label="Fine-tuning starts")
plt.title("Transfer Learning: Feature Extraction -> Fine-Tuning")
plt.xlabel("Epoch")
plt.ylabel("Accuracy")
plt.legend()
plt.grid(True)
plt.show()

EXPERIMENT 8: Object Detection using YOLOv5 (PyTorch Hub)
--------------------------------------------------------------------------------
import torch
import cv2
import matplotlib.pyplot as plt
import urllib.request
import numpy as np

# 1. Download a sample image if not present
sample_url = "https://raw.githubusercontent.com/ultralytics/yolov5/master/data/images/zidane.jpg"
image_path = "zidane.jpg"
urllib.request.urlretrieve(sample_url, image_path)

# 2. Load pre-trained YOLOv5s model from PyTorch Hub
model = torch.hub.load("ultralytics/yolov5", "yolov5s", pretrained=True)

# 3. Perform Inference
results = model(image_path)

# 4. Print detections to console (class, confidence, bbox)
results.print()
detections_df = results.pandas().xyxy[0]
print("\nDetected Bounding Boxes:\n", detections_df[["name", "confidence", "xmin", "ymin", "xmax", "ymax"]])

# 5. Render detections on the image
rendered_imgs = results.render() # Returns a list with rendered images in BGR format
output_image = rendered_imgs[0]

# Convert BGR to RGB for matplotlib display
plt.figure(figsize=(10, 6))
plt.imshow(cv2.cvtColor(output_image, cv2.COLOR_BGR2RGB))
plt.axis("off")
plt.title("YOLOv5 Real-time Object Detection")
plt.savefig("yolo_detection_output.png")
plt.show()
