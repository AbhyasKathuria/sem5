Q10. Neo4j Student Course
CREATE
(:Student {id:1,name:'Rahul'}),
(:Student {id:2,name:'Arun'}),
(:Student {id:3,name:'Ram'}),
(:Student {id:4,name:'Kumar'}),
(:Student {id:5,name:'Ravi'})

(:Course {id:101,name:'DBMS'}),
(:Course {id:102,name:'AI'}),
(:Course {id:103,name:'Python'}),
(:Course {id:104,name:'Java'}),
(:Course {id:105,name:'ML'})

CREATE (s)-[:ENROLLED_IN]->(c)
MATCH (s:Student {id:1}),(c:Course {id:102})
CREATE (s)-[:ENROLLED_IN]->(c)
MATCH (n)
RETURN n
